INDEPENDENT VENTURE • Multi-Agent System

E-Commerce Pipeline Architecture

An automated 4-stage Python pipeline that ingests raw designs, cleans background transparency, verifies print specs, prices for margin, and publishes live to the storefront.

Stage 1: Perimeter Flood-Fill & Halo Removal

  • Uses a Breadth-First Search (BFS) queue seeded along the outer canvas perimeter to flood-fill and strip backgrounds without accidentally deleting internal design white fills.
  • Runs an anti-aliasing kernel scanning 4-way neighboring pixels [(0,1), (1,0), (0,-1), (-1,0)] to eliminate blurry compression halos along dark contours.
  • Crops directly to the artwork bounding box with a 2% proportional safety margin for print alignment.
View Python Code: Perimeter BFS Flood-Fill image_processor.py 
# Stage 1: Perimeter BFS Flood-Fill & Anti-Aliasing Cleanup
from PIL import Image
from collections import deque

def clean_design_background(input_path):
    """Perimeter flood-fill: Removes background without erasing internal design elements."""
    img = Image.open(input_path).convert("RGBA")
    width, height = img.size
    pixels = img.load()
    
    visited, queue = set(), deque()
    # Seed perimeter pixels into the BFS queue
    for x in range(width):
        queue.extend([(x, 0), (x, height - 1)])
    for y in range(height):
        queue.extend([(0, y), (width - 1, y)])
        
    while queue:
        x, y = queue.popleft()
        if (x, y) in visited or not (0 <= x < width and 0 <= y < height):
            continue
        visited.add((x, y))
        
        r, g, b, a = pixels[x, y]
        if r > 180 and g > 180 and b > 180 and a > 0:
            pixels[x, y] = (0, 0, 0, 0)
            for dx, dy in [(0, 1), (1, 0), (0, -1), (-1, 0)]:
                queue.append((x + dx, y + dy))
                
    return img

# Stage 2: Automated Pre-Flight Quality Gate
from PIL import Image

def verify_print_quality(image_path, min_dim=1000):
    """Pre-flight QA: Validates transparency, canvas coverage, and resolution."""
    with Image.open(image_path) as img:
        img = img.convert("RGBA")
        
        # 1. Reject if design lacks transparent cutouts
        if img.getextrema()[3][0] == 255:
            return False, "QA Fail: Missing transparent background."
        
        # 2. Reject if artwork utilization is too low (excessive whitespace)
        bbox = img.getbbox()
        if bbox:
            w, h = img.size
            bw, bh = bbox[2] - bbox[0], bbox[3] - bbox[1]
            if bw < w * 0.85 and bh < h * 0.85:
                return False, f"QA Fail: Design utilization too low ({bw}x{bh})."
        
        # 3. Minimum resolution gate for physical merchandise
        if img.width < min_dim or img.height < min_dim:
            return False, f"QA Fail: Resolution below {min_dim}px minimum."
            
        return True, "QA Passed: Approved for publishing"

Stage 3: Landed-Cost Pricing & Margin Model

  • Shipping Cost Absorption: Explicitly integrates $4.00 standard tracked shipping into product landed cost to protect margins.
  • 30% Gross Margin Target: Applies a 1.30 multiplier across base manufacturing costs to maintain sustainable profit buffers.
  • Retail Psychology Rounding: Automatically maps calculated cents to consumer-friendly .49 or .99 endings with hard category price floors ($2.49 for stickers).

# Stage 3: Landed-Cost Pricing & Margin Engine
def calculate_retail_price(base_cost_map, product_type):
    """Calculates landed cost, absorbs shipping, and optimizes retail price points."""
    retail_prices = {}
    SHIPPING_CENTS = 400     # Absorb $4.00 standard tracked shipping
    TARGET_MARGIN = 1.30     # 30% gross margin target
    
    for variant_id, base_cost in base_cost_map.items():
        landed_cost = base_cost + SHIPPING_CENTS
        calculated = int(landed_cost * TARGET_MARGIN)
        
        # Psychological price rounding (.49 or .99)
        cents = calculated % 100
        base_dollars = (calculated // 100) * 100
        retail_cents = base_dollars + (49 if cents < 50 else 99)
        
        # Enforce category price floor
        if product_type in ("Die-Cut", "Kiss-Cut"):
            retail_cents = max(retail_cents, 249)  # $2.49 minimum floor
            
        retail_prices[variant_id] = retail_cents
    return retail_prices

import requests, base64, os

def publish_product(shop_id, image_path, title, tags, retail_prices):
    """Uploads artwork and drafts live listing via Printify API."""
    # 1. Base64 encode and upload to media library
    with open(image_path, "rb") as f:
        encoded = base64.b64encode(f.read()).decode("utf-8")
    upload_resp = requests.post(
        "https://api.printify.com/v1/uploads/images.json",
        headers=HEADERS,
        json={"file_name": os.path.basename(image_path), "contents": encoded}
    )
    image_id = upload_resp.json()["id"]
    
    # 2. Configure variants, print coordinates, and draft product
    payload = {
        "title": title, "tags": tags,
        "blueprint_id": 12, "print_provider_id": 29,  # Bella+Canvas 3001
        "variants": [{"id": vid, "price": p, "is_enabled": True} for vid, p in retail_prices.items()],
        "print_areas": [{"placeholders": [{"position": "front", "images": [{"id": image_id, "scale": 0.9}]}]}]
    }
    return requests.post(f"https://api.printify.com/v1/shops/{shop_id}/products.json", headers=HEADERS, json=payload)

-- Airline Route Market Share & HHI Analysis
-- Calculate market share per carrier on overlapping routes
CREATE OR REPLACE VIEW overlap_market_shares AS
SELECT origin, dest, unique_carrier_name, total_passengers,
    ROUND((total_passengers::numeric / SUM(total_passengers) OVER (PARTITION BY origin, dest)) * 100, 2) AS market_share
FROM overlap_route_passengers;

-- Pre-merger HHI calculation across overlapping routes
CREATE OR REPLACE VIEW pre_merger_hhi AS
SELECT origin, dest,
    ROUND(SUM(POWER(market_share, 2)), 0) AS hhi
FROM overlap_market_shares
GROUP BY origin, dest;

-- Post-merger HHI combining JetBlue (B6) and Spirit (NK)
CREATE OR REPLACE VIEW post_merger_hhi AS
SELECT origin, dest,
    ROUND(SUM(POWER(market_share, 2)), 0) AS post_hhi
FROM post_merger_market_shares
GROUP BY origin, dest;

-- Flag routes exceeding DOJ antitrust presumption (Post-HHI > 2500 & Delta >= 200)
SELECT p.origin, p.dest, p.hhi AS pre_hhi, m.post_hhi, (m.post_hhi - p.hhi) AS delta_hhi
FROM pre_merger_hhi p
JOIN post_merger_hhi m ON p.origin = m.origin AND p.dest = m.dest
WHERE m.post_hhi > 2500 AND (m.post_hhi - p.hhi) >= 200
ORDER BY delta_hhi DESC;

# US Census Bureau API Batch Ingestion & Demographic Pipeline
import pandas as pd
import requests

CENSUS_API_URL = "https://api.census.gov/data/2022/acs/acs5"
CENSUS_VARS = {"B19013_001E": "median_income", "B01003_001E": "total_population"}

def fetch_census_batch(zip_codes, api_key, batch_size=50):
    all_data = []
    for i in range(0, len(zip_codes), batch_size):
        batch = zip_codes[i:i + batch_size]
        params = {
            "get": f"NAME,{','.join(CENSUS_VARS.keys())}",
            "for": f"zip code tabulation area:{','.join(batch)}",
            "key": api_key
        }
        resp = requests.get(CENSUS_API_URL, params=params)
        if resp.status_code == 200:
            raw = resp.json()
            df_batch = pd.DataFrame(raw[1:], columns=raw[0])
            all_data.append(df_batch)
    return pd.concat(all_data, ignore_index=True)

Airbnb Pricing & Demand Modeling

Analyzed multi-city Airbnb listing data in R to evaluate price elasticity of demand and model optimal nightly pricing across metropolitan markets.

What I Did

  • Filtered listing data to city-specific interquartile ranges (IQR) to clean out extreme pricing outliers across five major metros.
  • Estimated linear demand functions for each metropolitan market in R using ordinary least squares regression.
  • Calculated own-price elasticity of demand (OPE) at median prices (finding Los Angeles at -1.05) to advise hosts on revenue-maximizing pricing strategies.

# Airbnb Demand Schedule & Elasticity in R
library(dplyr)
library(ggplot2)

metros <- c("Austin", "Chicago", "DC", "LA", "Minneapolis")
results <- data.frame()

for (m in metros) {
  metro_data <- airbnb_2019 %>% filter(metro == m)
  p25 <- quantile(metro_data$price, 0.25)
  p75 <- quantile(metro_data$price, 0.75)
  
  demand <- metro_data %>%
    mutate(bookings = number_of_reviews_ltm * (2/4)) %>%
    mutate(nights_booked = if_else(bookings * 3 > 63, 63, bookings * 3)) %>%
    filter(price >= p25, price <= p75) %>%
    group_by(price) %>%
    summarize(demand = sum(nights_booked), .groups = "drop")
  
  # Demand regression: Q = a + b*P
  model <- lm(demand ~ price, data = demand)
  dQ_dP <- coef(model)[2]
  P <- median(metro_data$price)
  Q <- coef(model)[1] + coef(model)[2] * P
  
  # Own-Price Elasticity of Demand (OPE) at median price
  OPE <- dQ_dP * P / Q
  results <- rbind(results, data.frame(metro = m, OPE = OPE))
}
