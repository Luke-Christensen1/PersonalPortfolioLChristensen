Antitrust Economics & Pricing Power: Modeled route market concentration across 10,000+ US domestic origin-destination pairs for the JetBlue / Spirit proposed merger. Used SQL window functions (OVER (PARTITION BY origin, dest)) to calculate dynamic market shares and Delta HHI, identifying routes breaching the DOJ antitrust presumption threshold (>2,500 HHI and ΔHHI ≥ 200). 
-- JetBlue / Spirit Merger Antitrust Model: DOJ HHI Concentration
-- Evaluates route market power and monopoly pricing risk across 10k+ pairs

WITH RoutePassengers AS (
    SELECT 
        origin, 
        dest, 
        carrier,
        SUM(passengers) AS carrier_passengers,
        SUM(SUM(passengers)) OVER (PARTITION BY origin, dest) AS total_route_passengers
    FROM us_dot_db1b_flights
    WHERE year = 2024
    GROUP BY origin, dest, carrier
),

CarrierMarketShare AS (
    SELECT 
        origin, 
        dest, 
        carrier,
        carrier_passengers,
        total_route_passengers,
        ROUND((carrier_passengers::numeric / total_route_passengers) * 100, 2) AS market_share_pct
    FROM RoutePassengers
),

PrePostMergerHHI AS (
    SELECT 
        origin,
        dest,
        total_route_passengers,
        -- Pre-Merger HHI (Sum of squared carrier market shares)
        ROUND(SUM(POWER(market_share_pct, 2)), 0) AS pre_hhi,
        -- Post-Merger HHI combining JetBlue (B6) + Spirit (NK) shares
        ROUND(SUM(
            POWER(
                CASE 
                    WHEN carrier IN ('B6', 'NK') THEN 
                        SUM(CASE WHEN carrier IN ('B6', 'NK') THEN market_share_pct ELSE 0 END) OVER (PARTITION BY origin, dest)
                    ELSE market_share_pct 
                END, 2
            )
        ), 0) AS post_hhi
    FROM CarrierMarketShare
    GROUP BY origin, dest, total_route_passengers
)

SELECT 
    origin,
    dest,
    pre_hhi,
    post_hhi,
    (post_hhi - pre_hhi) AS delta_hhi,
    CASE 
        WHEN post_hhi > 2500 AND (post_hhi - pre_hhi) >= 200 THEN 'HIGH ANTITRUST SCRUTINY (DOJ PRESUMPTION)'
        WHEN post_hhi BETWEEN 1500 AND 2500 AND (post_hhi - pre_hhi) >= 100 THEN 'MODERATE CONCENTRATION RISK'
        ELSE 'UNLIKELY ADVERSE COMPETITIVE IMPACT'
    END AS regulatory_risk_assessment
FROM PrePostMergerHHI
WHERE total_route_passengers > 50000
ORDER BY delta_hhi DESC;
Geospatial Expansion & TAM Pipeline: Automated web scrapers for Trader Joe's and Sprouts Farmers Market stores, ingested US Census Bureau 5-Year ACS API demographic data (household income, education, population density), and executed geodesic spatial joins to score high-potential expansion zip codes for executive Tableau dashboards. 
import requests
import pandas as pd
from geopy.distance import geodesic

# 1. Ingest US Census Bureau 5-Year ACS Demographic Profiles
CENSUS_API_URL = "https://api.census.gov/data/2022/acs/acs5"
ACS_VARIABLES = {
    "B19013_001E": "median_household_income",
    "B01003_001E": "total_population",
    "B15003_022E": "bachelors_degree_count"
}

def fetch_census_demographics(api_key: str) -> pd.DataFrame:
    """Fetches zip-code level income, population, and education profiles."""
    params = {
        "get": f"NAME,{','.join(ACS_VARIABLES.keys())}",
        "for": "zip code tabulation area:*",
        "key": api_key
    }
    response = requests.get(CENSUS_API_URL, params=params).json()
    df = pd.DataFrame(response[1:], columns=response[0]).rename(columns=ACS_VARIABLES)
    df["median_household_income"] = pd.to_numeric(df["median_household_income"], errors="coerce")
    return df.dropna(subset=["median_household_income"])

# 2. Spatial Geodesic Join & Expansion Opportunity Scoring
def score_candidate_locations(candidates: pd.DataFrame, competitors: pd.DataFrame, census_df: pd.DataFrame):
    """Calculates 3-mile trade radius overlap and unserved market opportunity."""
    scored_sites = []
    for _, candidate in candidates.iterrows():
        cand_coord = (candidate["lat"], candidate["lon"])
        
        # Count competitor saturation (Whole Foods, Trader Joe's) within 3.0 miles
        comp_count = sum(
            geodesic(cand_coord, (c["lat"], c["lon"])).miles <= 3.0 
            for _, c in competitors.iterrows()
        )
        
        # Merge census income for candidate zip code
        zip_data = census_df.loc[census_df["zip"] == candidate["zip_code"]]
        med_income = zip_data["median_household_income"].values[0] if not zip_data.empty else 0
        
        # Opportunity Index: High disposable income / Low direct organic saturation
        index = (med_income / 10000) * (1.0 / (comp_count + 1))
        scored_sites.append({**candidate, "comp_count_3mi": comp_count, "opportunity_index": round(index, 2)})
        
    return pd.DataFrame(scored_sites).sort_values(by="opportunity_index", ascending=False)

# Specification: log(bookings) = beta_0 + beta_1 * log(price) + gamma * X + alpha_metro + eps
airbnb_clean <- read_csv("data/airbnb_multicity_raw.csv") %>%
  filter(price > 15 & price < 2500, estimated_bookings_30d > 0) %>%
  mutate(
    log_bookings = log(estimated_bookings_30d),
    log_price = log(price),
    host_is_superhost = as.factor(host_is_superhost),
    room_type = as.factor(room_type)
  )

# 2. Multivariate Regression with Neighborhood Fixed Effects
elasticity_model <- feols(
  log_bookings ~ log_price + accommodates + bedrooms + review_scores_rating + 
                 instant_bookable | metro_market^neighbourhood_cleansed,
